  const handleDownloadScript = () => {
    const isReverse = direction === 'DIN_TO_LOC';

import {
  SPECIAL_WORDS,
  COMBI_AIU,
  SPECIALS,
  PREPOSITIONS_AL,
  COMBI_LONG_VOWELS,
  ARTICLE_AL,
  ALPHABET_MAPPING,
  FINAL_ADJUSTMENTS,
  GENERIC_ARTICLE_CONTRACTION,
  POST_PROCESSING_CORRECTIONS,
  SUN_LETTERS_MAP
} from './mappings';

export const convertLocToDin = (input: string): string => {
  if (!input) return "";
  let text = " " + input.trim() + " ";
  ALPHABET_MAPPING.forEach(([key, value]) => { text = text.split(key).join(value); });
  SPECIAL_WORDS.forEach(([key, value]) => { text = text.split(key).join(value); });
  text = text.replace(/([ʿʾ]?(?!Rasūl)\p{Lu}\S*)\s+Allāh/gu, "$1allāh");
  FINAL_ADJUSTMENTS.forEach(([key, value]) => { text = text.split(key).join(value); });
  if (text.endsWith(" al- ")) {
    const firstWordMatch = text.trim().match(/^[^\s,]+/);
    if (firstWordMatch) {
      const firstChar = firstWordMatch[0][0].toLowerCase();
      const assimilation = SUN_LETTERS_MAP[firstChar];
      if (assimilation) { text = text.substring(0, text.length - 5) + " " + assimilation + " "; }
    }
  }
  SPECIALS.forEach(([key, value]) => { text = text.split(key).join(value); });
  text = text.replace(/((?:[Aa]l|[Ll]il|[Bb]il|[Kk]al|[Ww]al|bi-al-|ka-al-|wa-al-|[ʿʾ][Aa]l)-\S+ī)(\s+|-)([Aa]l-)/g, (match, p1, p2, p3) => {
    return p1 + p2 + "@@@" + p3 + "@@@";
  });
  COMBI_AIU.forEach(([key, value]) => { text = text.split(key).join(value); });
  POST_PROCESSING_CORRECTIONS.forEach(([key, value]) => { text = text.split(key).join(value); });
  PREPOSITIONS_AL.forEach(([key, value]) => { text = text.split(key).join(value); });
  COMBI_LONG_VOWELS.forEach(([key, value]) => { text = text.split(key).join(value); });
  GENERIC_ARTICLE_CONTRACTION.forEach(([key, value]) => { text = text.split(key).join(value); });
  ARTICLE_AL.forEach(([key, value]) => { text = text.split(key).join(value); });
  text = text.split("@@@al-@@@").join("al-");
  text = text.split("@@@Al-@@@").join("Al-");
  text = text.replace(/(?<![Aa]ll\u0101|ll\u0101)ah(?![a-z\u00C0-\u017F\u1E00-\u1EFF\-])/gi, "a");
  return text.trim();
};

export const convertDinToLoc = (input: string): string => {
  if (!input) return "";
  let text = " " + input.trim() + " ";
  text = text.replace(/(\S+)allāh/g, "$1 Allāh");
  const applyReverse = (mapping: string[][]) => { [...mapping].reverse().forEach(([locKey, dinValue]) => { text = text.split(dinValue).join(locKey); }); };
  applyReverse(ARTICLE_AL);
  text = text.replace(/ (ad|at|aḏ|az|aṯ|ar|as|aš|aṣ|aḍ|aṭ|aẓ|an)-/g, " al-");
  text = text.replace(/-(ad|at|aḏ|az|aṯ|ar|as|aš|aṣ|aḍ|aṭ|aẓ|an)-/g, "-al-");
  applyReverse(GENERIC_ARTICLE_CONTRACTION);
  applyReverse(COMBI_LONG_VOWELS);
  applyReverse(PREPOSITIONS_AL);
  applyReverse(POST_PROCESSING_CORRECTIONS);
  const reversibleSpecials = SPECIALS.filter(([k]) => k !== "á");
  applyReverse(reversibleSpecials);
  applyReverse(COMBI_AIU);
  applyReverse(FINAL_ADJUSTMENTS);
  applyReverse(SPECIAL_WORDS);
  applyReverse(ALPHABET_MAPPING);
  text = text.replace(/ (Ibn|ibn) /g, " ibn ");
  text = text.replace(/^ (ibn|Ibn) /g, " Ibn ");
  text = text.replace(/, (ibn|Ibn) /g, ", Ibn ");
  text = text.replace(/(?<!\swa)a(?![a-z\u00C0-\u017F\u1E00-\u1EFF\-])/gi, "ah");
  return text.trim();
};

import { PERSIAN_DIN_TO_LOC, PERSIAN_LOC_TO_DIN } from './persianMappings';
import { SPECIAL_WORDS, FINAL_ADJUSTMENTS, ARTICLE_AL, COMBI_LONG_VOWELS, SUN_LETTERS_MAP } from './mappings';

export const convertPersianDinToLoc = (input: string): string => {
  if (!input) return "";
  let text = " " + input.trim() + " ";
  text = text.replace(/-nāma(?![a-z\u00C0-\u017F\u1E00-\u1EFF])/g, "'nāmah");
  text = text.replace(/(\w)a-i(?![a-z\u00C0-\u017F\u1E00-\u1EFF])/g, "$1ah-'i");
  text = text.replace(/(\w)a(?![a-z\u00C0-\u017F\u1E00-\u1EFF])/g, '$1ah');
  const applyReverse = (mapping: string[][]) => { [...mapping].reverse().forEach(([locKey, dinValue]) => { text = text.split(dinValue).join(locKey); }); };
  applyReverse(FINAL_ADJUSTMENTS);
  applyReverse(SPECIAL_WORDS);
  applyReverse(ARTICLE_AL);
  text = text.replace(/ (ad|at|aḏ|az|aṯ|ar|as|aš|aṣ|aḍ|aṭ|aẓ|an)-/g, " al-");
  text = text.replace(/-(ad|at|aḏ|az|aṯ|ar|as|aš|aṣ|aḍ|aṭ|aẓ|an)-/g, "-al-");
  applyReverse(COMBI_LONG_VOWELS);
  PERSIAN_DIN_TO_LOC.forEach(([key, value]) => { text = text.split(key).join(value); });
  text = text.replace(/ (Ibn|ibn) /g, " ibn ");
  text = text.replace(/^ (ibn|Ibn) /g, " Ibn ");
  text = text.replace(/, (ibn|Ibn) /g, ", Ibn ");
  return text.trim();
};

export const convertPersianLocToDin = (input: string): string => {
  if (!input) return "";
  let text = " " + input.trim() + " ";
  text = text.replace(/['’]nāmah(?![a-z\u00C0-\u017F\u1E00-\u1EFF])/g, '-nāma');
  text = text.replace(/ah-[']?i(?![a-z\u00C0-\u017F\u1E00-\u1EFF])/g, 'a-i');
  text = text.replace(/ah(?![a-z\u00C0-\u017F\u1E00-\u1EFF])/g, 'a');
  PERSIAN_LOC_TO_DIN.forEach(([key, value]) => {
    if (key === "aw" || key === "ay" || key === "Aw" || key === "Ay") {
      const regex = new RegExp(`${key}(?![a\u0101i\u012Bu\u016B\u00E1])`, 'g');
      text = text.replace(regex, value);
    } else {
      text = text.split(key).join(value);
    }
  });
  COMBI_LONG_VOWELS.forEach(([key, value]) => { text = text.split(key).join(value); });
  FINAL_ADJUSTMENTS.forEach(([key, value]) => { text = text.split(key).join(value); });
  SPECIAL_WORDS.forEach(([key, value]) => { text = text.split(key).join(value); });
  ARTICLE_AL.forEach(([key, value]) => { text = text.split(key).join(value); });
  if (text.endsWith(" al- ")) {
    const firstWordMatch = text.trim().match(/^[^\s,]+/);
    if (firstWordMatch) {
      let firstChar = firstWordMatch[0][0].toLowerCase();
      const twoChars = firstWordMatch[0].substring(0, 2).toLowerCase();
      const mapping = PERSIAN_LOC_TO_DIN.find(([k]) => k === twoChars || k === firstChar);
      if (mapping) { firstChar = mapping[1][0]; }
      const assimilation = SUN_LETTERS_MAP[firstChar];
      if (assimilation) { text = text.substring(0, text.length - 5) + " " + assimilation + " "; }
    }
  }
  return text.trim();
};

const TRANS_TO_ARABIC_MAP: Record<string, string> = {
  'b': 'ب', 't': 'ت', 'ṯ': 'ث', 'ǧ': 'ج', 'ḥ': 'ح', 'ḫ': 'خ', 'd': 'د', 'ḏ': 'ذ', 'r': 'ر', 'z': 'ز', 's': 'س', 'š': 'ش', 'ṣ': 'ص', 'ḍ': 'ض', 'ṭ': 'ط', 'ẓ': 'ظ', 'ʿ': 'ع', 'ġ': 'غ', 'f': 'ف', 'q': 'ق', 'k': 'ك', 'l': 'ل', 'm': 'م', 'n': 'ن', 'h': 'ه', 'w': 'و', 'y': 'ي', 'ʾ': 'ء', 'ā': 'ا', 'ū': 'و', 'ī': 'ي', 'th': 'ث', 'kh': 'خ', 'dh': 'ذ', 'sh': 'ش', 'gh': 'غ'
};
const PUNCTUATION_MAP: Record<string, string> = { ',': '،', ';': '؛', '?': '؟' };
export const convertToArabic = (trans: string): string => {
  if (!trans) return "";
  let text = trans.trim().toLowerCase();
  text = text.replace(/allāh/g, " الله");
  text = text.replace(/(^|\s|-)a[tṯdḏrzsšṣḍṭẓln]-(?=[^\s])/g, "$1 ال");
  text = text.replace(/(^|\s|-)al-(?=[^\s])/g, "$1 ال");
  text = text.replace(/ a[tṯdḏrzsšṣḍṭẓln]- $/g, " الـ ");
  text = text.replace(/ al- $/g, " الـ ");
  text = text.replace(/ibn-?/g, " ابن ");
  text = text.replace(/bint-?/g, " بنت ");
  text = text.replace(/abū-?/g, " أبو ");
  text = text.replace(/ʿabd-?/g, " عبد ");
  text = text.replace(/(^|\s)([aiu])/g, "$1ا");
  let result = "";
  for (let i = 0; i < text.length; i++) {
    if (i < text.length - 1) {
      const digraph = text.substring(i, i + 2);
      if (TRANS_TO_ARABIC_MAP[digraph]) { result += TRANS_TO_ARABIC_MAP[digraph]; i++; continue; }
    }
    const char = text[i];
    if (TRANS_TO_ARABIC_MAP[char]) { result += TRANS_TO_ARABIC_MAP[char]; }
    else if (PUNCTUATION_MAP[char]) { result += PUNCTUATION_MAP[char]; }
    else if (/\s/.test(char) || char === '-' || char === ',' || char === '.') { result += char; }
  }
  return result.replace(/\s+/g, ' ').replace(/ال /g, 'ال').replace(/ - /g, '-').trim();
};
